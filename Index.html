function connectWallet(){ try { const tc = new TON_CONNECT_UI({ url: 'https://tonkeeper.com', manifestUrl: 'https://cdn.jsdelivr.net/gh/stillinog/Tuman@main/.well-known/tonconnect.json' }); tc.connect(); tc.onStatusChange(s => { if (s) { document.getElementById('wallet').innerHTML = '' + s.address.slice(0, 6) + '...' + s.address.slice(-4) + ''; } }); } catch (e) { console.error('TonConnect init error:', e); } }
